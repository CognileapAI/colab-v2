```sql
WITH ext AS (SELECT dataset_id, lower(coalesce(substring(file_name FROM '\.([^.]+)$'), '')) AS e
             FROM d3_file WHERE kind = '본체')
SELECT dataset_id, count(DISTINCT e) AS ext_kinds, array_agg(DISTINCT e ORDER BY e) AS exts
FROM ext GROUP BY dataset_id HAVING count(DISTINCT e) > 1 ORDER BY dataset_id;
```
